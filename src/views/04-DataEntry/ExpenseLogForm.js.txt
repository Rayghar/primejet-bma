// src/views/04-DataEntry/ExpenseLogForm.js
import React, { useEffect, useMemo, useState } from 'react';
import { logExpense } from '../../api/dataEntryService';
import { EXPENSE_CATEGORIES } from '../../utils/constants';
import Button from '../../components/shared/Button';
import { PlusCircle } from 'lucide-react';

const parseMoney = (v) => {
  const s = String(v ?? '').replace(/,/g, '').trim();
  if (!s) return NaN;
  const n = Number(s);
  return Number.isFinite(n) ? n : NaN;
};

const safeId = (o) => o?.id || o?._id;

export default function ExpenseLogForm({ user, plants = [], onSuccess, onError }) {
  const defaultBranchId = useMemo(() => (plants.length ? safeId(plants[0]) : ''), [plants]);

  const initialState = useMemo(
    () => ({
      description: '',
      amount: '',
      category: EXPENSE_CATEGORIES?.[0] || 'Miscellaneous',
      branchId: defaultBranchId,
      date: new Date().toISOString().split('T')[0],
    }),
    [defaultBranchId]
  );

  const [formData, setFormData] = useState(initialState);
  const [isSubmitting, setIsSubmitting] = useState(false);

  // Keep branchId synced when plants load (don’t clobber if user already selected)
  useEffect(() => {
    if (plants.length > 0 && !formData.branchId) {
      setFormData((prev) => ({ ...prev, branchId: defaultBranchId }));
    }
  }, [plants, defaultBranchId, formData.branchId]);

  // If initialState changes (plants loaded), reset only if form is still pristine
  useEffect(() => {
    // avoid wiping in-progress typing
    const pristine =
      !formData.description && !formData.amount && formData.category === (EXPENSE_CATEGORIES?.[0] || 'Miscellaneous');
    if (pristine) setFormData(initialState);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [initialState]);

  const handleChange = (e) => {
    const { name, value } = e.target;
    setFormData((prev) => ({ ...prev, [name]: value }));
  };

  const canSubmit = useMemo(() => {
    const amt = parseMoney(formData.amount);
    return (
      !!formData.date &&
      !!formData.branchId &&
      !!formData.category &&
      !!formData.description.trim() &&
      Number.isFinite(amt) &&
      amt > 0
    );
  }, [formData]);

  const handleSubmit = async (e) => {
    e.preventDefault();
    if (isSubmitting) return;

    if (!canSubmit) {
      onError?.('Please fill in all required fields with valid values.');
      return;
    }

    setIsSubmitting(true);
    try {
      const amount = parseMoney(formData.amount);

      await logExpense({
        description: formData.description.trim(),
        category: formData.category,
        branchId: formData.branchId,
        date: formData.date,
        amount,
        submittedBy: user?.email,
      });

      onSuccess?.('Expense logged successfully!');
      setFormData(initialState);
    } catch (error) {
      console.error('Expense Log Error:', error);
      onError?.(error.response?.data?.message || 'Failed to submit expense log. Please try again.');
    } finally {
      setIsSubmitting(false);
    }
  };

  return (
    <form onSubmit={handleSubmit} className="space-y-4">
      <div>
        <label className="block text-sm font-medium text-gray-700">Date</label>
        <input
          type="date"
          name="date"
          value={formData.date}
          onChange={handleChange}
          className="mt-1 block w-full p-2 border rounded-md"
          required
        />
      </div>

      <div>
        <label className="block text-sm font-medium text-gray-700">Branch</label>
        <select
          name="branchId"
          value={formData.branchId}
          onChange={handleChange}
          className="mt-1 block w-full p-2 border rounded-md bg-white"
          required
        >
          {plants.length > 0 ? (
            plants.map((b) => (
              <option key={safeId(b)} value={safeId(b)}>
                {b.name}
              </option>
            ))
          ) : (
            <option value="">Loading branches...</option>
          )}
        </select>
      </div>

      <div>
        <label className="block text-sm font-medium text-gray-700">Category</label>
        <select
          name="category"
          value={formData.category}
          onChange={handleChange}
          className="mt-1 block w-full p-2 border rounded-md bg-white"
          required
        >
          {(EXPENSE_CATEGORIES || []).map((cat) => (
            <option key={cat}>{cat}</option>
          ))}
        </select>
      </div>

      <div>
        <label className="block text-sm font-medium text-gray-700">Description</label>
        <input
          type="text"
          name="description"
          value={formData.description}
          onChange={handleChange}
          className="mt-1 block w-full p-2 border rounded-md"
          placeholder="e.g., Generator fuel"
          required
        />
      </div>

      <div>
        <label className="block text-sm font-medium text-gray-700">Amount (₦)</label>
        <input
          type="text"
          inputMode="decimal"
          name="amount"
          value={formData.amount}
          onChange={handleChange}
          className="mt-1 block w-full p-2 border rounded-md"
          placeholder="e.g., 50,000"
          required
        />
      </div>

      <div className="flex justify-end pt-2">
        <Button type="submit" disabled={!canSubmit || isSubmitting} icon={PlusCircle}>
          {isSubmitting ? 'Submitting...' : 'Add Expense'}
        </Button>
      </div>
    </form>
  );
}
