// src/views/04-DataEntry/EndOfDayReportModal.js
import React, { useEffect, useMemo, useState } from 'react';
import { getDailySummaryReport } from '../../api/dataEntryService';
import { formatDate, formatCurrency } from '../../utils/formatters';
import Modal from '../../components/shared/Modal';
import Button from '../../components/shared/Button';
import { Printer, AlertTriangle } from 'lucide-react';

const safeArr = (v) => (Array.isArray(v) ? v : []);
const safeNum = (v, d = 0) => {
  const n = Number(v);
  return Number.isFinite(n) ? n : d;
};

export default function EndOfDayReportModal({ summaryId, onClose }) {
  const [reportData, setReportData] = useState(null);
  const [loading, setLoading] = useState(true);
  const [err, setErr] = useState('');

  useEffect(() => {
    const fetchReportData = async () => {
      if (!summaryId) {
        setReportData(null);
        setLoading(false);
        return;
      }

      setLoading(true);
      setErr('');
      try {
        const data = await getDailySummaryReport(summaryId);
        setReportData(data);
      } catch (error) {
        console.error('Failed to fetch report data:', error);
        setErr(error?.response?.data?.message || 'Failed to generate report.');
        setReportData(null);
      } finally {
        setLoading(false);
      }
    };

    fetchReportData();
  }, [summaryId]);

  const titleDate = useMemo(() => {
    const d = reportData?.summary?.date || new Date();
    return formatDate(d);
  }, [reportData]);

  const handlePrint = () => {
    if (!reportData) return;

    // Prevent HTML injection in printed output
    const esc = (s) =>
      String(s ?? '')
        .replaceAll('&', '&amp;')
        .replaceAll('<', '&lt;')
        .replaceAll('>', '&gt;')
        .replaceAll('"', '&quot;')
        .replaceAll("'", '&#039;');

    const printWindow = window.open('', '', 'height=700,width=900');
    if (!printWindow) return;

    const summary = reportData.summary || {};
    const totals = reportData.totals || {};
    const sales = safeArr(reportData.sales);
    const expenses = safeArr(reportData.expenses);

    printWindow.document.write('<html><head><title>Print Report</title>');
    printWindow.document.write('<style>');
    printWindow.document.write(`
      body { font-family: system-ui, -apple-system, Segoe UI, Roboto, sans-serif; margin: 20px; color: #111; }
      .report-header { text-align: center; margin-bottom: 20px; }
      .report-header h1 { font-size: 22px; margin: 0 0 4px 0; }
      .report-header h2 { font-size: 16px; margin: 0 0 10px 0; font-weight: 600; }
      .muted { color: #555; font-size: 12px; }
      .section { margin-top: 16px; }
      .section h3 { font-size: 14px; margin: 0 0 8px 0; border-bottom: 1px solid #eee; padding-bottom: 6px; }
      .row { display: flex; justify-content: space-between; gap: 12px; padding: 6px 0; }
      .row strong { font-weight: 700; }
      .hr { border-top: 1px dashed #ccc; margin: 14px 0; }
      table { width: 100%; border-collapse: collapse; margin-top: 6px; }
      th, td { text-align: left; font-size: 12px; padding: 8px; border-bottom: 1px solid #eee; }
      th { background: #f7f7f7; }
      .right { text-align: right; }
      @media print { .no-print { display: none !important; } }
    `);
    printWindow.document.write('</style></head><body>');

    const reportContent = `
      <div class="report-header">
        <h1>PrimeJet Gas LLC</h1>
        <h2>End-of-Day Summary</h2>
        <div class="muted">Date: ${esc(formatDate(summary.date))}</div>
        <div class="muted">Branch: ${esc(summary.branchName || summary.branchId)}</div>
        <div class="muted">Cashier: ${esc(summary.cashierName || 'N/A')}</div>
      </div>

      <div class="hr"></div>

      <div class="section">
        <h3>Summary</h3>
        <div class="row"><span>Total Revenue</span><strong>${esc(formatCurrency(totals.totalRevenue))}</strong></div>
        <div class="row"><span>Total LPG Sold</span><strong>${esc(safeNum(totals.totalKgSold).toLocaleString())} kg</strong></div>
        <div class="row"><span>Total Expenses</span><strong>${esc(formatCurrency(totals.totalExpenses))}</strong></div>
      </div>

      <div class="hr"></div>

      <div class="section">
        <h3>Sales Details</h3>
        ${
          sales.length
            ? `<table>
                <thead><tr><th>KG</th><th>Method</th><th class="right">Amount</th></tr></thead>
                <tbody>
                  ${sales
                    .map(
                      (s) =>
                        `<tr>
                          <td>${esc(safeNum(s.kgSold).toFixed(2))}</td>
                          <td>${esc(s.transactionType || s.paymentMethod || '-')}</td>
                          <td class="right">${esc(formatCurrency(s.amount || s.revenue))}</td>
                        </tr>`
                    )
                    .join('')}
                </tbody>
              </table>`
            : '<div class="muted">No sales recorded.</div>'
        }
      </div>

      <div class="hr"></div>

      <div class="section">
        <h3>Expense Details</h3>
        ${
          expenses.length
            ? `<table>
                <thead><tr><th>Description</th><th>Category</th><th class="right">Amount</th></tr></thead>
                <tbody>
                  ${expenses
                    .map(
                      (e) =>
                        `<tr>
                          <td>${esc(e.description || '-')}</td>
                          <td>${esc(e.category || '-')}</td>
                          <td class="right">${esc(formatCurrency(e.amount))}</td>
                        </tr>`
                    )
                    .join('')}
                </tbody>
              </table>`
            : '<div class="muted">No expenses recorded.</div>'
        }
      </div>

      <div class="hr"></div>

      <div class="section">
        <h3>Meter Readings</h3>
        <div class="row"><span>Opening Meter A</span><strong>${esc(safeNum(summary?.openingMeters?.meterA).toFixed(2))} kg</strong></div>
        <div class="row"><span>Closing Meter A</span><strong>${esc(safeNum(summary?.closingMeters?.meterA).toFixed(2))} kg</strong></div>
        <div class="row"><span>Opening Meter B</span><strong>${esc(safeNum(summary?.openingMeters?.meterB).toFixed(2))} kg</strong></div>
        <div class="row"><span>Closing Meter B</span><strong>${esc(safeNum(summary?.closingMeters?.meterB).toFixed(2))} kg</strong></div>
        <div class="row"><span>Price Per Kg</span><strong>${esc(formatCurrency(summary.pricePerKg))}</strong></div>
      </div>
    `;

    printWindow.document.write(reportContent);
    printWindow.document.write('</body></html>');
    printWindow.document.close();
    printWindow.focus();
    printWindow.print();
    printWindow.close();
  };

  return (
    <Modal title={`End-of-Day Report for ${titleDate}`} onClose={onClose}>
      {loading ? (
        <p className="text-center p-8">Generating report...</p>
      ) : err ? (
        <div className="p-6 rounded-lg bg-red-500/10 border border-red-500/20 text-red-200 text-sm flex items-start gap-2">
          <AlertTriangle size={16} className="mt-0.5 text-red-300" />
          <div>{err}</div>
        </div>
      ) : reportData ? (
        <div>
          <div id="report-content" className="space-y-4">
            <div className="report-header text-center">
              <h1 className="text-xl font-bold text-white">PrimeJet Gas LLC</h1>
              <h2 className="text-sm text-gray-300 font-semibold">End-of-Day Summary</h2>
              <p className="text-xs text-gray-400">Date: {formatDate(reportData.summary.date)}</p>
              <p className="text-xs text-gray-400">Branch: {reportData.summary.branchName || reportData.summary.branchId}</p>
              <p className="text-xs text-gray-400">Cashier: {reportData.summary.cashierName || 'N/A'}</p>
            </div>

            <div className="border-t border-white/10" />

            <div className="summary-section">
              <h3 className="font-bold text-white mb-2">Summary</h3>
              <div className="space-y-2 text-sm">
                <div className="flex justify-between">
                  <span className="text-gray-400">Total Revenue</span>
                  <span className="text-white font-bold">{formatCurrency(reportData.totals.totalRevenue)}</span>
                </div>
                <div className="flex justify-between">
                  <span className="text-gray-400">Total LPG Sold</span>
                  <span className="text-white font-bold">{safeNum(reportData.totals.totalKgSold).toLocaleString()} kg</span>
                </div>
                <div className="flex justify-between">
                  <span className="text-gray-400">Total Expenses</span>
                  <span className="text-white font-bold">{formatCurrency(reportData.totals.totalExpenses)}</span>
                </div>
              </div>
            </div>

            <div className="border-t border-white/10" />

            <div className="summary-section">
              <h3 className="font-bold text-white mb-2">Sales Details</h3>
              {reportData.sales?.length ? (
                <div className="space-y-2 text-sm">
                  {reportData.sales.map((s, i) => (
                    <div key={`sale-${s._id || i}`} className="flex justify-between bg-white/5 border border-white/10 rounded-lg p-2">
                      <span className="text-gray-300">
                        {safeNum(s.kgSold).toFixed(2)} kg ({s.transactionType || s.paymentMethod || '—'})
                      </span>
                      <span className="text-white font-bold">{formatCurrency(s.amount || s.revenue)}</span>
                    </div>
                  ))}
                </div>
              ) : (
                <p className="text-gray-500 text-sm">No sales recorded.</p>
              )}
            </div>

            <div className="border-t border-white/10" />

            <div className="summary-section">
              <h3 className="font-bold text-white mb-2">Expense Details</h3>
              {reportData.expenses?.length ? (
                <div className="space-y-2 text-sm">
                  {reportData.expenses.map((e, i) => (
                    <div key={`exp-${e._id || i}`} className="flex justify-between bg-white/5 border border-white/10 rounded-lg p-2">
                      <span className="text-gray-300">
                        {e.description} ({e.category})
                      </span>
                      <span className="text-white font-bold">{formatCurrency(e.amount)}</span>
                    </div>
                  ))}
                </div>
              ) : (
                <p className="text-gray-500 text-sm">No expenses recorded.</p>
              )}
            </div>

            <div className="border-t border-white/10" />

            <div className="summary-section">
              <h3 className="font-bold text-white mb-2">Meter Readings</h3>
              <div className="grid grid-cols-1 md:grid-cols-2 gap-2 text-sm">
                <div className="bg-white/5 border border-white/10 rounded-lg p-2 flex justify-between">
                  <span className="text-gray-400">Opening Meter A</span>
                  <span className="text-white font-bold">{safeNum(reportData.summary.openingMeters?.meterA).toFixed(2)} kg</span>
                </div>
                <div className="bg-white/5 border border-white/10 rounded-lg p-2 flex justify-between">
                  <span className="text-gray-400">Closing Meter A</span>
                  <span className="text-white font-bold">{safeNum(reportData.summary.closingMeters?.meterA).toFixed(2)} kg</span>
                </div>
                <div className="bg-white/5 border border-white/10 rounded-lg p-2 flex justify-between">
                  <span className="text-gray-400">Opening Meter B</span>
                  <span className="text-white font-bold">{safeNum(reportData.summary.openingMeters?.meterB).toFixed(2)} kg</span>
                </div>
                <div className="bg-white/5 border border-white/10 rounded-lg p-2 flex justify-between">
                  <span className="text-gray-400">Closing Meter B</span>
                  <span className="text-white font-bold">{safeNum(reportData.summary.closingMeters?.meterB).toFixed(2)} kg</span>
                </div>
                <div className="bg-white/5 border border-white/10 rounded-lg p-2 flex justify-between md:col-span-2">
                  <span className="text-gray-400">Price Per Kg</span>
                  <span className="text-white font-bold">{formatCurrency(reportData.summary.pricePerKg)}</span>
                </div>
              </div>
            </div>
          </div>

          <div className="flex justify-end pt-4 no-print">
            <Button onClick={handlePrint} icon={Printer}>
              Print Report
            </Button>
          </div>
        </div>
      ) : (
        <p className="text-center p-8 text-gray-500">
          Could not generate report. Summary not found or data is incomplete.
        </p>
      )}
    </Modal>
  );
}
