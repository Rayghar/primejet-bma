// src/views/04-DataEntry/TransactionHistory.js
import React, { useEffect, useMemo, useState } from 'react';
import { getTransactionHistory } from '../../api/dataEntryService';
import PageTitle from '../../components/shared/PageTitle';
import Card from '../../components/shared/Card';
import { formatCurrency, formatDate } from '../../utils/formatters';
import { Search, RefreshCw, Filter } from 'lucide-react';

const safeArr = (v) => (Array.isArray(v) ? v : []);
const safeNum = (v, d = 0) => {
  const n = Number(v);
  return Number.isFinite(n) ? n : d;
};

export default function TransactionHistory() {
  const [transactions, setTransactions] = useState([]);
  const [loading, setLoading] = useState(true);

  const [q, setQ] = useState('');
  const [type, setType] = useState('all'); // all | sale | expense

  const fetchHistory = async () => {
    setLoading(true);
    try {
      const data = await getTransactionHistory();
      setTransactions(safeArr(data?.transactions || data));
    } catch (e) {
      console.error('Transaction History Error', e);
      setTransactions([]);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchHistory();
  }, []);

  const filtered = useMemo(() => {
    const query = String(q || '').toLowerCase();

    return transactions.filter((t) => {
      const desc = String(t.description || '').toLowerCase();
      const ttype = String(t.type || '').toLowerCase();
      const cashier = String(t.cashierName || t.submittedBy || '').toLowerCase();

      const matchesText = !query || desc.includes(query) || ttype.includes(query) || cashier.includes(query);
      const matchesType = type === 'all' || ttype === type;

      return matchesText && matchesType;
    });
  }, [transactions, q, type]);

  return (
    <div className="space-y-6">
      <div className="flex justify-between items-center">
        <PageTitle title="Transaction Ledger" subtitle="Historical record of all POS entries" />

        <div className="flex items-center gap-2">
          <div className="relative">
            <input
              type="text"
              placeholder="Search ledger..."
              className="glass-input pl-10 py-2 w-64 text-sm"
              value={q}
              onChange={(e) => setQ(e.target.value)}
            />
            <Search className="absolute left-3 top-2.5 text-gray-500" size={16} />
          </div>

          <div className="relative">
            <select
              value={type}
              onChange={(e) => setType(e.target.value)}
              className="glass-input py-2 pl-9 pr-8 text-sm bg-black/20"
              title="Filter by type"
            >
              <option value="all">All</option>
              <option value="sale">Sales</option>
              <option value="expense">Expenses</option>
            </select>
            <Filter className="absolute left-3 top-2.5 text-gray-500" size={16} />
          </div>

          <button onClick={fetchHistory} className="glass-button px-3 py-2 flex items-center text-sm" title="Refresh">
            <RefreshCw size={16} className="mr-2" />
            Refresh
          </button>
        </div>
      </div>

      <Card className="p-0 overflow-hidden">
        <div className="overflow-x-auto">
          <table className="w-full text-left text-sm text-gray-400">
            <thead className="bg-white/5 text-gray-300 uppercase text-xs">
              <tr>
                <th className="p-4">Date</th>
                <th className="p-4">Type</th>
                <th className="p-4">Description</th>
                <th className="p-4 text-right">Amount</th>
                <th className="p-4">Cashier</th>
              </tr>
            </thead>

            <tbody className="divide-y divide-white/5">
              {loading ? (
                <tr>
                  <td className="p-6 text-center text-gray-500" colSpan={5}>
                    Loading transactions…
                  </td>
                </tr>
              ) : filtered.length === 0 ? (
                <tr>
                  <td className="p-6 text-center text-gray-500" colSpan={5}>
                    No transactions found.
                  </td>
                </tr>
              ) : (
                filtered.map((tx) => (
                  <tr key={tx.id || tx._id} className="hover:bg-white/5">
                    <td className="p-4">{formatDate(tx.date || tx.createdAt)}</td>

                    <td className="p-4">
                      <span
                        className={`px-2 py-1 rounded text-[10px] uppercase font-bold ${
                          tx.type === 'sale'
                            ? 'bg-green-500/20 text-green-400'
                            : 'bg-red-500/20 text-red-400'
                        }`}
                      >
                        {tx.type}
                      </span>
                    </td>

                    <td className="p-4 text-white">
                      {tx.type === 'sale' ? `${safeNum(tx.kgSold).toFixed(2)}kg LPG` : tx.description || '—'}
                    </td>

                    <td
                      className={`p-4 text-right font-mono font-bold ${
                        tx.type === 'sale' ? 'text-green-400' : 'text-red-400'
                      }`}
                    >
                      {tx.type === 'expense' ? '-' : ''}
                      {formatCurrency(safeNum(tx.amount, safeNum(tx.revenue, 0)))}
                    </td>

                    <td className="p-4 text-xs">{tx.cashierName || tx.submittedBy || 'Unknown'}</td>
                  </tr>
                ))
              )}
            </tbody>
          </table>
        </div>
      </Card>
    </div>
  );
}
