import React, { useMemo, useState, useEffect } from 'react';
import {
  getInventorySummary,
  getStockInHistory,
  getCylinders,
  addCylinder,
  deleteCylinder,
  addStockIn,
} from '../../api/inventoryService';

import PageTitle from '../../components/shared/PageTitle';
import Card from '../../components/shared/Card';
import Button from '../../components/shared/Button';
import Modal from '../../components/shared/Modal';
import {
  Package,
  AlertTriangle,
  TrendingUp,
  PlusCircle,
  Trash2,
  Factory,
  RefreshCw,
  Download,
} from 'lucide-react';

const safeNum = (v, d = 0) => {
  const n = Number(v);
  return Number.isFinite(n) ? n : d;
};

const getId = (x) => x?.id || x?._id;

const toYMD = (d) => {
  try {
    const dt = d ? new Date(d) : new Date();
    if (Number.isNaN(dt.getTime())) return new Date().toISOString().split('T')[0];
    return dt.toISOString().split('T')[0];
  } catch {
    return new Date().toISOString().split('T')[0];
  }
};

const formatDate = (d) => {
  try {
    const dt = new Date(d);
    if (Number.isNaN(dt.getTime())) return '—';
    return dt.toLocaleDateString();
  } catch {
    return '—';
  }
};

const exportToCsv = (filename, rows) => {
  if (!rows || rows.length === 0) return;

  const escape = (val) => {
    const s = String(val ?? '');
    if (s.includes(',') || s.includes('"') || s.includes('\n')) {
      return `"${s.replaceAll('"', '""')}"`;
    }
    return s;
  };

  const headers = Object.keys(rows[0]);
  const csv = [
    headers.map(escape).join(','),
    ...rows.map((r) => headers.map((h) => escape(r[h])).join(',')),
  ].join('\n');

  const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
  const url = URL.createObjectURL(blob);

  const link = document.createElement('a');
  link.href = url;
  link.setAttribute('download', filename);
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
  URL.revokeObjectURL(url);
};

export default function Inventory() {
  const [summary, setSummary] = useState(null);
  const [history, setHistory] = useState([]);
  const [cylinders, setCylinders] = useState([]);

  const [loading, setLoading] = useState(true);
  const [syncError, setSyncError] = useState('');

  // Modal states
  const [showStockModal, setShowStockModal] = useState(false);
  const [showCylinderModal, setShowCylinderModal] = useState(false);

  const refreshData = async () => {
    setLoading(true);
    setSyncError('');
    try {
      const [sum, hist, cyl] = await Promise.all([
        getInventorySummary(),
        getStockInHistory(),
        getCylinders(),
      ]);

      setSummary(sum || null);
      setHistory(Array.isArray(hist) ? hist : []);
      setCylinders(Array.isArray(cyl) ? cyl : []);
    } catch (e) {
      console.error('Inventory Sync Failed', e);
      setSyncError(e?.response?.data?.message || e?.message || 'Inventory sync failed.');
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    refreshData();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  const totalCapacity = useMemo(() => safeNum(summary?.totalCapacity, 0), [summary]);
  const currentStock = useMemo(() => safeNum(summary?.currentStock, 0), [summary]);

  const stockPercent = useMemo(() => {
    if (totalCapacity <= 0) return 0;
    const pct = (currentStock / totalCapacity) * 100;
    if (!Number.isFinite(pct)) return 0;
    return Math.max(0, Math.min(100, pct));
  }, [currentStock, totalCapacity]);

  const totalCylinders = useMemo(() => {
    return cylinders.reduce((sum, c) => sum + safeNum(c?.quantity, 0), 0);
  }, [cylinders]);

  const handleRemoveCylinder = async (id) => {
    if (!id) return;
    if (!window.confirm('Delete this cylinder batch?')) return;

    try {
      await deleteCylinder(id);
      await refreshData();
    } catch (e) {
      console.error('Delete cylinder failed', e);
      setSyncError(e?.response?.data?.message || e?.message || 'Failed to delete cylinder batch.');
    }
  };

  const handleExportHistory = () => {
    const rows = (history || []).map((b) => ({
      PurchaseDate: formatDate(b.purchaseDate),
      Supplier: b.supplier || '',
      QuantityKg: safeNum(b.quantityKg, 0),
      RemainingKg: safeNum(b.remainingKg, 0),
      CostPerKg: safeNum(b.costPerKg, 0),
      TotalCost: safeNum(b.totalCost, safeNum(b.quantityKg, 0) * safeNum(b.costPerKg, 0)),
      EstimatedRevenue: safeNum(b.estimatedActualRevenue, 0),
      ProfitLoss: safeNum(b.profitLoss, 0),
      ProfitMarginPct: safeNum(b.profitMargin, 0),
      SalesProgressPct: safeNum(b.salesProgress, 0),
    }));

    exportToCsv(`stockin_profitability_${toYMD(new Date())}.csv`, rows);
  };

  if (loading) {
    return (
      <div className="p-8 text-center text-blue-400 animate-pulse">
        Loading Inventory Intelligence...
      </div>
    );
  }

  return (
    <div className="space-y-6">
      <div className="flex justify-between items-center">
        <PageTitle title="Inventory Intelligence" subtitle="Stock Levels & Batch Profitability" />
        <div className="flex gap-2">
          <Button onClick={refreshData} variant="secondary" icon={RefreshCw}>
            Refresh
          </Button>
          <Button onClick={() => setShowCylinderModal(true)} variant="secondary" icon={Package}>
            Cylinders
          </Button>
          <Button onClick={() => setShowStockModal(true)} icon={PlusCircle}>
            Log Stock-In
          </Button>
        </div>
      </div>

      {syncError ? (
        <div className="bg-red-500/10 border border-red-500/20 p-4 rounded-xl text-red-200 text-sm">
          <div className="flex items-start">
            <AlertTriangle className="text-red-400 mr-3 mt-0.5" size={18} />
            <div>
              <p className="font-bold">Inventory Sync Error</p>
              <p className="text-xs mt-1 opacity-90">{syncError}</p>
            </div>
          </div>
        </div>
      ) : null}

      {/* TOP ROW */}
      <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
        {/* Gauge */}
        <div className="glass-card flex flex-col items-center justify-center relative overflow-hidden">
          <div className="absolute inset-0 bg-blue-600/5"></div>
          <h3 className="font-bold text-gray-400 mb-6 uppercase tracking-wider text-xs z-10">
            Current LPG Level
          </h3>

          <div className="relative w-48 h-48 flex items-center justify-center">
            <svg className="w-full h-full transform -rotate-90">
              <circle cx="96" cy="96" r="88" className="stroke-white/5" strokeWidth="12" fill="none" />
              <circle
                cx="96"
                cy="96"
                r="88"
                className={`${stockPercent < 20 ? 'stroke-red-500' : 'stroke-blue-500'} transition-all duration-1000`}
                strokeWidth="12"
                fill="none"
                strokeDasharray={553}
                strokeDashoffset={553 - (553 * stockPercent) / 100}
                strokeLinecap="round"
              />
            </svg>
            <div className="absolute text-center">
              <span className="text-4xl font-bold text-white block">{currentStock.toLocaleString()}</span>
              <span className="text-xs text-gray-400 uppercase">Kilograms</span>
            </div>
          </div>

          <p className="text-xs text-gray-500 mt-4 z-10">
            Total Capacity: {totalCapacity.toLocaleString()} kg
          </p>
        </div>

        {/* Alerts + Cylinders */}
        <div className="flex flex-col gap-6">
          <div className="glass-card flex-1 flex flex-col justify-center">
            <h3 className="font-bold text-gray-400 mb-4 uppercase tracking-wider text-xs">
              System Alerts
            </h3>

            {stockPercent < 20 ? (
              <div className="bg-red-500/10 border border-red-500/20 p-4 rounded-xl flex items-start">
                <AlertTriangle className="text-red-500 mr-3 mt-0.5" size={20} />
                <div>
                  <h4 className="text-red-400 font-bold text-sm">Low Stock Warning</h4>
                  <p className="text-red-300/70 text-xs mt-1">
                    Reserves are critical. Re-order immediately.
                  </p>
                </div>
              </div>
            ) : (
              <div className="bg-green-500/10 border border-green-500/20 p-4 rounded-xl flex items-start">
                <Factory className="text-green-500 mr-3 mt-0.5" size={20} />
                <div>
                  <h4 className="text-green-400 font-bold text-sm">Optimal Levels</h4>
                  <p className="text-green-300/70 text-xs mt-1">Operations running normally.</p>
                </div>
              </div>
            )}
          </div>

          {/* Cylinder Summary */}
          <div className="glass-card flex-1">
            <div className="flex justify-between items-center mb-4">
              <h3 className="font-bold text-gray-400 uppercase tracking-wider text-xs">
                Cylinder Assets
              </h3>
              <span className="text-xs text-blue-400">{totalCylinders} Total</span>
            </div>

            <div className="space-y-2 max-h-32 overflow-y-auto pr-2">
              {cylinders.length === 0 ? (
                <div className="text-xs text-gray-500 p-2 bg-white/5 rounded-lg">
                  No cylinder batches recorded yet.
                </div>
              ) : (
                cylinders.map((c) => {
                  const id = getId(c);
                  return (
                    <div key={id} className="flex justify-between text-sm p-2 bg-white/5 rounded-lg">
                      <span className="text-gray-300">{c.size || '—'}</span>
                      <div className="flex items-center gap-3">
                        <span className="font-mono font-bold text-white">{safeNum(c.quantity, 0)}</span>
                        <button
                          onClick={() => handleRemoveCylinder(id)}
                          className="text-red-400 hover:text-red-300"
                          title="Delete"
                        >
                          <Trash2 size={14} />
                        </button>
                      </div>
                    </div>
                  );
                })
              )}
            </div>
          </div>
        </div>
      </div>

      {/* BOTTOM ROW: Profitability Table */}
      <div className="glass-card">
        <div className="flex items-center justify-between mb-6">
          <h3 className="font-bold text-white flex items-center">
            <TrendingUp className="mr-2 text-green-500" size={20} /> Batch Profitability Analysis
          </h3>
          <Button variant="secondary" size="sm" icon={Download} onClick={handleExportHistory}>
            Export CSV
          </Button>
        </div>

        <div className="overflow-x-auto">
          <table className="w-full text-left text-sm text-gray-400">
            <thead className="text-xs uppercase bg-white/5 text-gray-300 font-semibold">
              <tr>
                <th className="p-4 rounded-tl-xl">Date</th>
                <th className="p-4">Supplier</th>
                <th className="p-4 text-right">Cost</th>
                <th className="p-4 text-right">Est. Revenue</th>
                <th className="p-4 text-right">Net Profit</th>
                <th className="p-4 text-center">Margin</th>
                <th className="p-4 rounded-tr-xl">Progress</th>
              </tr>
            </thead>
            <tbody className="divide-y divide-white/5">
              {history.length === 0 ? (
                <tr>
                  <td colSpan="7" className="p-8 text-center text-gray-500">
                    No stock history available.
                  </td>
                </tr>
              ) : (
                history.map((batch) => {
                  const id = getId(batch) || `${batch.purchaseDate}-${batch.supplier}`;
                  const totalCost =
                    safeNum(batch.totalCost, safeNum(batch.quantityKg, 0) * safeNum(batch.costPerKg, 0));
                  const estRev = safeNum(batch.estimatedActualRevenue, 0);
                  const profitLoss = safeNum(batch.profitLoss, 0);
                  const profitMargin = safeNum(batch.profitMargin, 0);
                  const salesProgress = Math.max(0, Math.min(100, safeNum(batch.salesProgress, 0)));

                  return (
                    <tr key={id} className="hover:bg-white/5 transition-colors">
                      <td className="p-4 text-white">{formatDate(batch.purchaseDate)}</td>
                      <td className="p-4">{batch.supplier || '—'}</td>
                      <td className="p-4 text-right">₦{totalCost.toLocaleString()}</td>
                      <td className="p-4 text-right text-gray-300">₦{estRev.toLocaleString()}</td>
                      <td className={`p-4 text-right font-bold ${profitLoss >= 0 ? 'text-green-400' : 'text-red-400'}`}>
                        {profitLoss >= 0 ? '+' : ''}
                        {profitLoss.toLocaleString()}
                      </td>
                      <td className="p-4 text-center">
                        <span
                          className={`px-2 py-1 rounded text-xs font-bold ${
                            profitMargin > 15 ? 'bg-green-500/20 text-green-400' : 'bg-yellow-500/20 text-yellow-400'
                          }`}
                        >
                          {profitMargin}%
                        </span>
                      </td>
                      <td className="p-4">
                        <div className="flex items-center space-x-2">
                          <div className="w-16 bg-gray-700 h-1.5 rounded-full overflow-hidden">
                            <div className="bg-blue-500 h-full" style={{ width: `${salesProgress}%` }}></div>
                          </div>
                          <span className="text-[10px]">{salesProgress}%</span>
                        </div>
                      </td>
                    </tr>
                  );
                })
              )}
            </tbody>
          </table>
        </div>
      </div>

      {/* Modals */}
      {showStockModal && <AddStockModal onClose={() => setShowStockModal(false)} onRefresh={refreshData} />}
      {showCylinderModal && <AddCylinderModal onClose={() => setShowCylinderModal(false)} onRefresh={refreshData} />}
    </div>
  );
}

// --- Modals ---
const AddStockModal = ({ onClose, onRefresh }) => {
  const [form, setForm] = useState({
    quantityKg: '',
    costPerKg: '',
    supplier: '',
    purchaseDate: new Date().toISOString().split('T')[0],
    targetSalePricePerKg: '',
  });

  const [submitting, setSubmitting] = useState(false);
  const [err, setErr] = useState('');

  const handleSubmit = async (e) => {
    e.preventDefault();
    setErr('');

    const qty = safeNum(form.quantityKg, NaN);
    const cpk = safeNum(form.costPerKg, NaN);
    const tsp = safeNum(form.targetSalePricePerKg, NaN);

    if (!Number.isFinite(qty) || qty <= 0) return setErr('Quantity must be a positive number.');
    if (!Number.isFinite(cpk) || cpk <= 0) return setErr('Cost per KG must be a positive number.');
    if (!form.supplier?.trim()) return setErr('Supplier is required.');
    if (!form.purchaseDate) return setErr('Purchase date is required.');
    // targetSalePricePerKg is optional depending on your backend; keep it if supported:
    if (form.targetSalePricePerKg && (!Number.isFinite(tsp) || tsp <= 0)) {
      return setErr('Target sale price per KG must be a positive number.');
    }

    setSubmitting(true);
    try {
      await addStockIn({
        quantityKg: qty,
        costPerKg: cpk,
        supplier: form.supplier.trim(),
        purchaseDate: form.purchaseDate,
        ...(form.targetSalePricePerKg ? { targetSalePricePerKg: tsp } : {}),
      });

      await onRefresh();
      onClose();
    } catch (e2) {
      console.error('Stock-in failed', e2);
      setErr(e2?.response?.data?.message || e2?.message || 'Failed to log stock-in.');
    } finally {
      setSubmitting(false);
    }
  };

  return (
    <Modal title="Log Bulk Stock-In" onClose={onClose}>
      <form onSubmit={handleSubmit} className="space-y-4">
        {err ? <div className="text-xs text-red-300 bg-red-500/10 border border-red-500/20 p-2 rounded">{err}</div> : null}

        <div>
          <label className="block text-gray-400 text-xs mb-1">Quantity (KG)</label>
          <input
            type="number"
            className="glass-input w-full p-2"
            value={form.quantityKg}
            onChange={(e) => setForm({ ...form, quantityKg: e.target.value })}
            required
          />
        </div>

        <div className="grid grid-cols-2 gap-3">
          <div>
            <label className="block text-gray-400 text-xs mb-1">Cost per KG (₦)</label>
            <input
              type="number"
              className="glass-input w-full p-2"
              value={form.costPerKg}
              onChange={(e) => setForm({ ...form, costPerKg: e.target.value })}
              required
            />
          </div>

          <div>
            <label className="block text-gray-400 text-xs mb-1">Target Sale Price / KG (₦)</label>
            <input
              type="number"
              className="glass-input w-full p-2"
              value={form.targetSalePricePerKg}
              onChange={(e) => setForm({ ...form, targetSalePricePerKg: e.target.value })}
              placeholder="Optional"
            />
          </div>
        </div>

        <div>
          <label className="block text-gray-400 text-xs mb-1">Supplier</label>
          <input
            type="text"
            className="glass-input w-full p-2"
            value={form.supplier}
            onChange={(e) => setForm({ ...form, supplier: e.target.value })}
            required
          />
        </div>

        <div>
          <label className="block text-gray-400 text-xs mb-1">Purchase Date</label>
          <input
            type="date"
            className="glass-input w-full p-2"
            value={form.purchaseDate}
            onChange={(e) => setForm({ ...form, purchaseDate: e.target.value })}
            required
          />
        </div>

        <div className="flex justify-end pt-4 gap-2">
          <Button type="button" variant="secondary" onClick={onClose} disabled={submitting}>
            Cancel
          </Button>
          <Button type="submit" className="glass-button" disabled={submitting}>
            {submitting ? 'Logging...' : 'Confirm Stock-In'}
          </Button>
        </div>
      </form>
    </Modal>
  );
};

const AddCylinderModal = ({ onClose, onRefresh }) => {
  const [form, setForm] = useState({ size: '12.5 kg', quantity: '' });
  const [submitting, setSubmitting] = useState(false);
  const [err, setErr] = useState('');

  const handleSubmit = async (e) => {
    e.preventDefault();
    setErr('');

    const qty = safeNum(form.quantity, NaN);
    if (!Number.isFinite(qty) || qty <= 0) return setErr('Quantity must be a positive number.');

    setSubmitting(true);
    try {
      await addCylinder({ size: form.size, quantity: qty });
      await onRefresh();
      onClose();
    } catch (e2) {
      console.error('Add cylinder failed', e2);
      setErr(e2?.response?.data?.message || e2?.message || 'Failed to add cylinder batch.');
    } finally {
      setSubmitting(false);
    }
  };

  return (
    <Modal title="Add Cylinder Assets" onClose={onClose}>
      <form onSubmit={handleSubmit} className="space-y-4">
        {err ? <div className="text-xs text-red-300 bg-red-500/10 border border-red-500/20 p-2 rounded">{err}</div> : null}

        <div>
          <label className="block text-gray-400 text-xs mb-1">Size</label>
          <select
            className="glass-input w-full p-2 bg-slate-800"
            value={form.size}
            onChange={(e) => setForm({ ...form, size: e.target.value })}
          >
            <option>3 kg</option>
            <option>5 kg</option>
            <option>6 kg</option>
            <option>12.5 kg</option>
            <option>25 kg</option>
            <option>50 kg</option>
          </select>
        </div>

        <div>
          <label className="block text-gray-400 text-xs mb-1">Quantity</label>
          <input
            type="number"
            className="glass-input w-full p-2"
            value={form.quantity}
            onChange={(e) => setForm({ ...form, quantity: e.target.value })}
            required
          />
        </div>

        <div className="flex justify-end pt-4 gap-2">
          <Button type="button" variant="secondary" onClick={onClose} disabled={submitting}>
            Cancel
          </Button>
          <Button type="submit" className="glass-button" disabled={submitting}>
            {submitting ? 'Saving...' : 'Add Batch'}
          </Button>
        </div>
      </form>
    </Modal>
  );
};
